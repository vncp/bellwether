    font=\scriptsize,
    stage/.style={draw, rounded corners=1.5pt, fill=black!4, inner sep=1.5pt,

  % ---- The loop.

  \node[stage] (att) at (2.95,2.95) {attribute\\[-1pt]$\Delta$, $I$};

  \node[stage] (gat) at (1.75,1.75) {gate\\[-1pt]$\Eq(a, a_0)$};

  \draw[flow] (rep) -- (att);
